##Exemplo 1
##############################################################
import numpy as np
from scipy import linalg

A = np.array([[1,1,1],[1,-2,3],[2,2,-1]])
B = np.array([[6],[6],[3]])
det_A = linalg.det(A)
print('det(A) = ', det_A)
print('\n')
Xsol = linalg.inv(A)@B
print('Solução = ', Xsol)

##Exemplo 5 
##############################################################
import numpy as np
from scipy import linalg

x = df_pub['TV']
y = df_pub['vendas']
soma_x2 = sum(x**2)
soma_x = sum(x)
n = len(x)
soma_xy = sum(x*y)
soma_y = sum(y)

A = np.array([[soma_x2,soma_x],[soma_x, n]])
B = np.array([[soma_xy],[soma_y]])

Xsol = linalg.solve(A,B)

a = Xsol[0][0]
b = Xsol[1][0]

print('y = %.2fx + %.2f'% (a,b))

##############################################################

valor = 180
print('x = %.2f -> y = %.2f' % (valor, a*valor + b))
