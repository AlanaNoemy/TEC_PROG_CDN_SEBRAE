import numpy as np
from scipy import linalg

A = np.array([[3,0,2],[9,1,7],[1,0,1]])
print('A = ', A)
print('tipo: ', A.shape)
print('\n')

det_A = linalg.det(A)
print('det(A) = ' , det_A)
print('\n')

inv_A = linalg.inv(A)
print('inv(A) = ', inv_A)
print('\n')

#Configuração para impressão com 2 dígitos na mantissa
#np.set_printoptions(formatter = {'float': '{: 0.2f}'.format})

#só para confirmar :)
print('A*inv(A) = ', A.dot(inv_A))
